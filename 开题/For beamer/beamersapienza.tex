% ===================== 1. 研究背景与问题 =====================
\footlinecolor{sintefgreen}
\begin{frame}{一、研究背景与问题}
\framesubtitle{漏洞披露规模扩大，自动化检测需求迫切}
\small
\begin{itemize}
\item \textbf{漏洞规模持续增长}：2026 年前两个季度新增公开 CVE 漏洞逾 \alert{35{,}000} 项，
      人工代码审计已难以匹配其增长速度。
\item \textbf{传统静态分析的局限}：其检测能力依赖预定义规则，
      规则覆盖不足导致\emph{漏报}，规则过于宽泛导致\emph{误报}，且难以给出可解释的判定依据。
\item \textbf{大语言模型的机遇与不足}：LLM 具备代码语义理解与推理潜力，
      但仍存在两项短板——\emph{缺乏软件安全领域知识}，且\emph{孤立函数不足以支撑安全判定}。
\end{itemize}
\vspace{0.2em}
\begin{colorblock}[black]{sinteflightgreen}{研究切入点}
构建基于 LLM 的\textbf{自主漏洞检测智能体}，以\emph{证据驱动}的方式完成函数级漏洞判定：
先取证、后结论，并使全过程可追溯、可复核。
\end{colorblock}
\end{frame}

% ===================== 2. 研究目标与关键难点 =====================
\begin{frame}{二、研究目标与关键难点}
\framesubtitle{一项总目标，两项关键技术挑战}
\small
\begin{block}{总目标}
面向 C/C++ 函数代码，输入函数源码及其所属代码仓库地址，
输出三项内容：\alert{漏洞判定结果} —— \alert{判定理由} —— \alert{分析过程记录}。
\end{block}
\begin{columns}[t]
\begin{column}{0.48\textwidth}
\begin{colorblock}[black]{sinteflightgreen}{难点一：LLM 缺乏安全领域知识}
即使前沿模型亦难以充分掌握软件安全知识。\\
\textbf{对策}：构建领域知识 \textbf{RAG} 模块，
基于已披露的 CVE 实例建立可检索的知识库，为检测提供历史相似漏洞模式。
\end{colorblock}
\end{column}
\begin{column}{0.48\textwidth}
\begin{colorblock}[white]{sinteflilla}{难点二：孤立函数难以支撑判定}
漏洞常与调用方、被调用方、全局变量等上下文相关，单函数信息不完备。\\
\textbf{对策}：赋予智能体\textbf{仓库按需读取}能力，
在证据不足时自主挖掘关联上下文。
\end{colorblock}
\end{column}
\end{columns}
\vspace{0.1em}
{\footnotesize \textbf{设计原则}：证据驱动——智能体先基于领域知识初步分析，证据不足时再按需调用仓库工具；检索与读取记录均在界面留痕。}
\end{frame}

% ===================== 3. 系统架构与技术路线 =====================
\begin{frame}{三、系统架构与技术路线}
\framesubtitle{四层解耦结构：界面 $\rightarrow$ 智能体 $\rightarrow$ 工具 $\rightarrow$ 数据}
\centering
\begin{tikzpicture}[
  node distance=0.3cm,
  box/.style={draw=black!30, rounded corners=2pt, align=center,
              font=\footnotesize, minimum width=0.9\textwidth,
              minimum height=0.82cm, inner sep=3pt}
]
\node[box, fill=sinteflightgreen] (ui)
  {表现层（界面）：输入函数与仓库地址、实时查看分析过程、接收检测报告};
\node[box, fill=sinteflilla, text=white, below=of ui] (agent)
  {智能体层（决策）：\ 推理 $\rightarrow$ 工具调用 $\rightarrow$ 观测结果 $\rightarrow$ 再推理，证据充分方作出判定};
\node[box, fill=sintefyellow, below=of agent] (tools)
  {工具层（能力）：\ ① 领域知识检索（RAG）\ \ $|$\ \ ② 仓库上下文挖掘（grep / find / 调用关系解析）};
\node[box, fill=maincolor, text=white, below=of tools] (data)
  {数据层（资源）：历史 CVE 漏洞案例库\ \ $|$\ \ 代码仓库版本快照缓存};
\end{tikzpicture}
\vspace{0.4em}

{\footnotesize 分层解耦使界面、决策、能力与资源各自独立开发与替换——例如更换底层模型时，其余三层无需改动。}
\end{frame}

% ===================== 4. 核心模块设计 =====================
\begin{frame}{四、核心模块设计}
\framesubtitle{围绕两项关键难点，设计两个核心模块}
\footnotesize
\begin{columns}[t]
\begin{column}{0.5\textwidth}
\begin{colorblock}[black]{sinteflightgreen}{模块一：领域知识库（RAG）}
基于提供的历史案例数据集，通过向量相似度检索相似漏洞模式，供 LLM 参考。\\
\textbf{知识来源}：成对函数样本（漏洞版 / 修复版）、\texttt{cve\_desc}（披露描述）、
\texttt{commit\_message}（修复说明）、\texttt{cwe}（漏洞类型）及项目元信息。\\
\textbf{约束}：\emph{红线}——仅可使用所提供数据集，不得引入额外安全知识。
\end{colorblock}
\end{column}
\begin{column}{0.5\textwidth}
\begin{colorblock}[white]{sintefpurple}{模块二：代码上下文挖掘}
基于修复提交链接获取对应版本仓库快照，为智能体配备内容查找与读取工具，按需溯源。\\
\textbf{可挖掘上下文}：调用方与被调用方、全局变量声明与使用、类型与宏定义、相关辅助函数。\\
\textbf{实现}：GitHub API 获取快照，grep / find 检索，Tree-sitter 解析函数定义与调用关系。\\
\textbf{约束}：严格版本控制——漏洞版仅取父提交版本，修复版仅取提交后版本，且不提供 \texttt{git diff}。
\end{colorblock}
\end{column}
\end{columns}
\end{frame}

\item \textbf{阶段三（W6--W7）评测交付}：300 条样本批量评测与迭代调优；撰写技术报告、完成系统演示与结果提交。
